import { createRouteHandlerClient } from '@supabase/auth-helpers-nextjs'
import { cookies } from 'next/headers'
import { NextResponse } from 'next/server'

export async function POST(request: Request) {
  try {
    const { email } = await request.json()
    
    if (!email) {
      return NextResponse.json(
        { error: 'Email is required' },
        { status: 400 }
      )
    }

    const supabase = createRouteHandlerClient({ cookies })
    
    // Get the current session to verify admin privileges
    const { data: { session } } = await supabase.auth.getSession()
    
    if (!session) {
      return NextResponse.json(
        { error: 'Unauthorized - No session found' },
        { status: 401 }
      )
    }

    // Check if user is admin (owner)
    const { data: userData, error: userError } = await supabase
      .from('users')
      .select('role')
      .eq('email', session.user.email)
      .single()

    if (userError || userData?.role !== 'owner') {
      return NextResponse.json(
        { error: 'Insufficient permissions - Admin access required' },
        { status: 403 }
      )
    }

    // Get the user from auth
    const { data: authUsers, error: authError } = await supabase.auth.admin.listUsers()
    
    if (authError) {
      console.error('Error listing users:', authError)
      return NextResponse.json(
        { error: 'Failed to get user from auth: ' + authError.message },
        { status: 500 }
      )
    }

    // Find the user by email
    const userToConfirm = authUsers.users.find((u: any) => u.email === email)
    
    if (!userToConfirm) {
      return NextResponse.json(
        { error: 'User not found in auth system' },
        { status: 404 }
      )
    }

    // Update the auth user's email confirmation
    const { data: updatedUser, error: updateError } = await supabase.auth.admin.updateUserById(
      userToConfirm.id,
      { 
        email_confirm: true,
        // Also update user metadata
        user_metadata: {
          ...userToConfirm.user_metadata,
          confirmed_at: new Date().toISOString(),
          confirmed_by: session.user.email
        }
      }
    )

    if (updateError) {
      console.error('Error confirming user in auth:', updateError)
      return NextResponse.json(
        { error: 'Failed to confirm user: ' + updateError.message },
        { status: 500 }
      )
    }

    // Also update your users table
    const { error: usersTableError } = await supabase
      .from('users')
      .update({
        confirmed_at: new Date().toISOString(),
        confirmed_by: session.user.email
      })
      .eq('email', email)

    if (usersTableError) {
      console.error('Error updating users table:', usersTableError)
      // Don't return error here, as auth was already updated
    }

    return NextResponse.json({ 
      success: true, 
      message: `User ${email} confirmed successfully`,
      data: { 
        email, 
        confirmed: true,
        confirmed_at: new Date().toISOString(),
        confirmed_by: session.user.email
      }
    })
  } catch (error: any) {
    console.error('Error in confirm-auth API:', error)
    return NextResponse.json(
      { error: error.message || 'Failed to confirm user' },
      { status: 500 }
    )
  }
}
